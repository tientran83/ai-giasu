<script>
    const sessionId = "session_" + Math.random().toString(36).substring(2, 9);

    async function sendMessage() {
        const input = document.getElementById("userInput");
        const chatBox = document.getElementById("chatBox");
        const text = input.value.trim();

        if (!text) return;

        // Thêm tin nhắn học sinh vào khung chat
        chatBox.innerHTML += `<div class="message user-message">${text}</div>`;
        input.value = "";
        chatBox.scrollTop = chatBox.scrollHeight;

        // Thêm thông báo "Đang suy nghĩ..."
        const loadingId = "loading-" + Date.now();
        chatBox.innerHTML += `<div class="message bot-message loading" id="${loadingId}">Gia sư đang suy nghĩ...</div>`;
        chatBox.scrollTop = chatBox.scrollHeight;

        try {
            const response = await fetch('/api/tutor/chat', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    session_id: sessionId,
                    subject: "Toán học",
                    grade_level: "Lớp 9",
                    question: text
                })
            });

            const data = await response.json();
            document.getElementById(loadingId).remove();

            // Tạo nội dung câu trả lời đẹp đẽ
            let hintsHTML = "<ul>";
            data.hints.forEach(h => {
                hintsHTML += `<li><b>Bước ${h.step_number}:</b> ${h.explanation}</li>`;
            });
            hintsHTML += "</ul>";

            const botReply = `
                <div class="message bot-message">
                    <div class="praise">🌟 ${data.praise_or_encouragement}</div>
                    <div class="concept">${data.concept_explanation}</div>
                    ${hintsHTML}
                    <div class="question">❓ ${data.guided_question}</div>
                </div>
            `;

            chatBox.innerHTML += botReply;
            chatBox.scrollTop = chatBox.scrollHeight;

        } catch (error) {
            document.getElementById(loadingId).remove();
            chatBox.innerHTML += `<div class="message bot-message" style="color:red;">Đã có lỗi xảy ra. Vui lòng thử lại!</div>`;
        }
    }

    function handleKeyPress(e) {
        if (e.key === 'Enter') sendMessage();
    }
</script>
